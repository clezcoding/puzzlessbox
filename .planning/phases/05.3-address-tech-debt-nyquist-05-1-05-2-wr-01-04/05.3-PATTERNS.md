async def resolve_mcp_auth(payload: MCPAuthRequest, request: Request) -> dict[str, str]:
    settings = get_settings()
    service_bearer = request.headers.get("X-Service-Bearer")
    expected = settings.SERVICE_BEARER_TOKEN
    if (

    # Verify owner_id exists
    exists = session.execute(
        text('SELECT 1 FROM "user" WHERE id = CAST(:oid AS uuid) LIMIT 1'),
        {"oid": owner_id_env},

    if not exists:
        logger.error(f"SERVICE_OWNER_ID {owner_id_env} does not exist in 'user' table. Aborting bootstrap.")
        return
