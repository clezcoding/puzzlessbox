def _normalize_times(starts_at: datetime | None, ends_at: datetime | None) -> tuple[datetime, datetime]:
    if starts_at is None and ends_at is None:
        start = datetime.now(timezone.utc)
        end = start + timedelta(hours=1)
    elif starts_at is not None and ends_at is None:
        start = starts_at
        end = start + timedelta(hours=1)
    elif starts_at is None and ends_at is not None:
        end = ends_at
        start = end - timedelta(hours=1)
    else:
        start = starts_at
        end = ends_at
    return start, end
